// GS Etsy Uploader: Cloudflare Worker
// Holds the Etsy key as secrets, connects to Doug's Etsy shop once with OAuth (PKCE),
// and turns upload packages (picked from Doug's computer in the browser) into Etsy listings.
//
// Bindings required:
//   KV namespace binding:  STORE
//   Secrets:               ETSY_KEYSTRING, ETSY_SHARED_SECRET
// Optional variable:       ALLOWED_USER_ID (locks the tool to one Etsy user; set automatically on first connect if empty)

const ETSY_API = 'https://api.etsy.com/v3';
const SCOPES = 'listings_r listings_w shops_r';

export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    try {
      if (url.pathname === '/') return html(PAGE);
      if (url.pathname === '/connect') return connect(url, env);
      if (url.pathname === '/callback') return callback(url, env);
      if (url.pathname.startsWith('/api/')) {
        const sess = await session(req, env);
        if (!sess) return json({ error: 'not_connected' }, 401);
        if (url.pathname === '/api/status') return json({ ok: true, shop_id: sess.shop_id, shop_name: sess.shop_name });
        if (url.pathname === '/api/taxonomy') return taxonomy(url, env);
        if (url.pathname === '/api/listing' && req.method === 'POST') return createListing(req, env, sess);
        if (url.pathname === '/api/disconnect' && req.method === 'POST') {
          await env.STORE.delete('tok');
          return json({ ok: true }, 200, { 'Set-Cookie': 'gs_etsy=; Path=/; Max-Age=0; HttpOnly; Secure; SameSite=Lax' });
        }
      }
      return new Response('Not found', { status: 404 });
    } catch (e) {
      return json({ error: 'server_error', detail: String(e && e.message || e) }, 500);
    }
  }
};

// ---------- OAuth ----------
function b64url(bytes) {
  let s = btoa(String.fromCharCode(...new Uint8Array(bytes)));
  return s.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}
function randomString(n = 48) {
  return b64url(crypto.getRandomValues(new Uint8Array(n)));
}
async function connect(url, env) {
  const verifier = randomString(48);
  const challenge = b64url(await crypto.subtle.digest('SHA-256', new TextEncoder().encode(verifier)));
  const state = randomString(16);
  await env.STORE.put('pkce:' + state, verifier, { expirationTtl: 900 });
  const redirect = url.origin + '/callback';
  const q = new URLSearchParams({
    response_type: 'code', client_id: env.ETSY_KEYSTRING, redirect_uri: redirect,
    scope: SCOPES, state, code_challenge: challenge, code_challenge_method: 'S256'
  });
  return Response.redirect('https://www.etsy.com/oauth/connect?' + q.toString(), 302);
}
async function callback(url, env) {
  const state = url.searchParams.get('state') || '';
  const code = url.searchParams.get('code');
  const verifier = await env.STORE.get('pkce:' + state);
  if (!code || !verifier) return html(msgPage('Connection did not finish. Go back and click Connect to Etsy again.'), 400);
  await env.STORE.delete('pkce:' + state);
  const r = await fetch(ETSY_API + '/public/oauth/token', {
    method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({ grant_type: 'authorization_code', client_id: env.ETSY_KEYSTRING,
      redirect_uri: url.origin + '/callback', code, code_verifier: verifier })
  });
  const t = await r.json();
  if (!r.ok) return html(msgPage('Etsy said no: ' + escapeHtml(JSON.stringify(t))), 400);
  const userId = String(t.access_token).split('.')[0];
  const owner = env.ALLOWED_USER_ID || await env.STORE.get('owner');
  if (owner && owner !== userId) return html(msgPage('This uploader is locked to a different Etsy account.'), 403);
  if (!owner) await env.STORE.put('owner', userId);
  // Find the shop for this user
  const shops = await etsyFetch(env, t.access_token, '/application/users/' + userId + '/shops');
  const shop = await shops.json();
  if (!shops.ok || !shop.shop_id) return html(msgPage('Connected, but no Etsy shop was found on this account.'), 400);
  const sid = randomString(32);
  await env.STORE.put('tok', JSON.stringify({
    access: t.access_token, refresh: t.refresh_token, exp: Date.now() + (t.expires_in - 120) * 1000,
    user_id: userId, shop_id: shop.shop_id, shop_name: shop.shop_name, sid
  }));
  return new Response(null, { status: 302, headers: {
    Location: '/', 'Set-Cookie': `gs_etsy=${sid}; Path=/; Max-Age=2592000; HttpOnly; Secure; SameSite=Lax` } });
}
async function session(req, env) {
  const cookie = req.headers.get('Cookie') || '';
  const m = cookie.match(/gs_etsy=([A-Za-z0-9_-]+)/);
  if (!m) return null;
  const raw = await env.STORE.get('tok');
  if (!raw) return null;
  const tok = JSON.parse(raw);
  if (tok.sid !== m[1]) return null;
  if (Date.now() > tok.exp) {
    const r = await fetch(ETSY_API + '/public/oauth/token', {
      method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({ grant_type: 'refresh_token', client_id: env.ETSY_KEYSTRING, refresh_token: tok.refresh })
    });
    const t = await r.json();
    if (!r.ok) { await env.STORE.delete('tok'); return null; }
    tok.access = t.access_token; tok.refresh = t.refresh_token || tok.refresh;
    tok.exp = Date.now() + (t.expires_in - 120) * 1000;
    await env.STORE.put('tok', JSON.stringify(tok));
  }
  return tok;
}
function etsyFetch(env, access, path, init = {}) {
  const headers = Object.assign({ 'x-api-key': env.ETSY_KEYSTRING + ':' + env.ETSY_SHARED_SECRET }, init.headers || {});
  if (access) headers['Authorization'] = 'Bearer ' + access;
  return fetch(ETSY_API + path, Object.assign({}, init, { headers }));
}

// ---------- Taxonomy search ----------
let TAX_CACHE = null;
async function taxonomy(url, env) {
  if (!TAX_CACHE) {
    const r = await etsyFetch(env, null, '/application/seller-taxonomy/nodes');
    const d = await r.json();
    const flat = [];
    (function walk(nodes, path) {
      for (const n of nodes || []) {
        const p = path.concat(n.name);
        flat.push({ id: n.id, path: p.join(' > ') });
        walk(n.children, p);
      }
    })(d.results, []);
    TAX_CACHE = flat;
  }
  const q = (url.searchParams.get('q') || '').toLowerCase();
  return json({ results: TAX_CACHE.filter(n => n.path.toLowerCase().includes(q)).slice(0, 60) });
}

async function createListing(req, env, sess) {
  const form = await req.formData();
  const meta = JSON.parse(form.get('meta'));
  const shop = sess.shop_id;
  const steps = [];

  // 1. Draft listing
  const body = new URLSearchParams();
  body.set('quantity', String(meta.quantity || 999));
  body.set('title', meta.title);
  body.set('description', meta.description);
  body.set('price', Number(meta.price).toFixed(2));
  body.set('who_made', 'i_did');
  body.set('when_made', meta.when_made || '2020_2026');
  body.set('is_supply', 'false');
  body.set('taxonomy_id', String(meta.taxonomy_id));
  body.set('type', 'download');
  body.set('should_auto_renew', 'false');
  body.set('is_taxable', 'true');
  (meta.tags || []).slice(0, 13).forEach(t => body.append('tags[]', t));
  if (meta.shop_section_id) body.set('shop_section_id', String(meta.shop_section_id));
  let r = await etsyFetch(env, sess.access, `/application/shops/${shop}/listings`, {
    method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body });
  let d = await r.json();
  if (!r.ok) return json({ ok: false, step: 'create', status: r.status, error: d }, 400);
  const listingId = d.listing_id;
  steps.push('draft ' + listingId);

  // 2. Images
  for (let i = 0; i < 10; i++) {
    const img = form.get('image' + i);
    if (!img) continue;
    const fd = new FormData();
    fd.append('image', img, img.name);
    fd.append('rank', String(i + 1));
    r = await etsyFetch(env, sess.access, `/application/shops/${shop}/listings/${listingId}/images`, { method: 'POST', body: fd });
    if (!r.ok) return json({ ok: false, listing_id: listingId, step: 'image' + i, status: r.status, error: await r.text() }, 400);
    steps.push('image ' + (i + 1));
  }

  // 3. Digital files
  for (let i = 0; i < 5; i++) {
    const f = form.get('file' + i);
    if (!f) continue;
    const fd = new FormData();
    fd.append('file', f, f.name);
    fd.append('name', f.name);
    fd.append('rank', String(i + 1));
    r = await etsyFetch(env, sess.access, `/application/shops/${shop}/listings/${listingId}/files`, { method: 'POST', body: fd });
    if (!r.ok) return json({ ok: false, listing_id: listingId, step: 'file' + i, status: r.status, error: await r.text() }, 400);
    steps.push('file ' + (i + 1));
  }

  // 4. Make sure it is a download listing, auto-renew off, and publish if asked
  const upd = new URLSearchParams({ type: 'download', should_auto_renew: 'false' });
  if (meta.publish) upd.set('state', 'active');
  r = await etsyFetch(env, sess.access, `/application/shops/${shop}/listings/${listingId}`, {
    method: 'PATCH', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: upd });
  if (!r.ok) return json({ ok: false, listing_id: listingId, step: 'update', status: r.status, error: await r.text() }, 400);
  steps.push(meta.publish ? 'published' : 'left as draft');
  return json({ ok: true, listing_id: listingId, sku: meta.sku, steps });
}

"""GS Etsy package builder.

For each eligible blueprint SKU it writes, into  out/GS Etsy Upload Packages/<SKU>/ :
  - "<SKU> Start Here.pdf"    the one-page guide that goes in every download (brings buyers back to GS)

def safe_file_name(name):
    # Etsy digital file names: 3 to 70 characters, letters, numbers, spaces, hyphens, underscores, periods
    base, ext = os.path.splitext(name)
    base = re.sub(r"[^A-Za-z0-9 ._-]", "", base)
    base = re.sub(r"\s+", " ", base).strip()
    return base[:70 - len(ext)].rstrip() + ext


def clean_title(t):
    t = re.sub(r'\s+', ' ', t).strip()
    return t


def short_name(title):
    n = re.sub(r'\bBlueprint\b', '', title)
    n = re.sub(r'^The\s+', '', n.strip())
    return re.sub(r'\s+', ' ', n).strip(' -')


def chapters_of(pdf_path):
    r = pypdf.PdfReader(pdf_path)
    out = []
    for x in r.outline:
        if isinstance(x, list):
            continue
        t = re.sub(r'\s+', ' ', x.title).strip()
        t = re.sub(r'^Chapter\s+\d+\s*[:.-]\s*', '', t)
        if not t or re.match(r'(?i)^(disclaimer|how to use this book|current platform sources|sources|the last word|closing)', t):
            continue
        out.append(t)
    return len(r.pages), out


def first_paragraph(pdf_path):
    r = pypdf.PdfReader(pdf_path)
    for i in range(2, min(6, len(r.pages))):
        txt = r.pages[i].extract_text() or ''
        if 'DISCLAIMER' in txt.upper()[:200]:
            continue
        lines = [l.strip() for l in txt.split('\n')[2:] if l.strip()]
        para = ' '.join(lines)
        para = re.sub(r'\s+', ' ', para)
        sents = re.split(r'(?<=[.!?])\s+', para)
        s = ' '.join(sents[:2]).strip()
        if len(s) > 60:
            return s
    return ''


def etsy_title(title):
    base = short_name(title)
    base = re.sub(r'\s+Business$', '', base)
    t = f'{base} Business Blueprint, Step by Step Startup Plan, Printable PDF and Editable Word Guide'
    if len(t) > 140:
        t = f'{base} Business Blueprint, Startup Plan, PDF and Editable Word'
    if len(t) > 140:
        t = f'{base} Business Blueprint PDF'[:140]
    for ch in '%:&+':  # Etsy allows each of these once at most
        if t.count(ch) > 1:
            t = t.replace(ch, ' ')
    return re.sub(r'\s+', ' ', t).strip()


def tags_for(meta):
    base = short_name(meta['title']).lower()
    cand = []
    words = [w for w in re.findall(r"[a-z0-9']+", base) if w not in {'the', 'and', 'a', 'of', 'to', 'for', 'on', 'in', 'your'}]
    if len(base) <= 20:
        cand.append(base)
    for two in ([' '.join(words[:2]), ' '.join(words[-2:])] if len(words) > 2 else []):
        if len(two) <= 20:
            cand.append(two)
    for w in (meta.get('secondary', ''), meta.get('category', '')):
        w = re.sub(r"[^a-z0-9 '-]", '', w.lower()).strip()
        if w and len(w) <= 20:
            cand.append(w)
    cand += ['business plan', 'startup guide', 'small business', 'side hustle', 'business blueprint',
             'start a business', 'business template', 'entrepreneur gift', 'digital download']
    out = []
    for c in cand:
        c = re.sub(r'\s+', ' ', c).strip()
        if c and c not in out and len(c) <= 20:
            out.append(c)
    return out[:13]


def unit(ch, pages=999):

import re, docx, pypdf
SKIP = re.compile(r'(?i)^(glamourless systems$|build your profit scaffold|20\d\d .*(update|gate)|vendor lock-in|important notice|disclaimer|how to use this|current (platform )?sources|sources|the last word|closing|a final word|table of contents|contents|about the author|copyright|legal|notice|appendix|glossary|index|references)')
NUMS = 'one two three four five six seven eight nine ten eleven twelve thirteen fourteen fifteen sixteen seventeen eighteen nineteen twenty twenty-one twenty-two twenty-three twenty-four twenty-five twenty-six twenty-seven twenty-eight twenty-nine thirty'.split()
WORDS = '(?:' + '|'.join(sorted(NUMS, key=len, reverse=True)) + ')'
def _num(x):
    return int(x) if x.isdigit() else NUMS.index(x.lower()) + 1
def _clean(t):
    t = re.sub(r'\s+', ' ', t).strip()
    t = re.sub(r'^(?:chapter|part|section)\s+(?:\d+[A-Za-z]?|' + WORDS + r')\b\s*[:.\-–—]?\s*', '', t, flags=re.I).strip(' :.-–—')
    t = re.sub(r'\s+\d{1,3}$', '', t).strip()
    if t.isupper():
        t = t.title().replace("'S ", "'s ").replace("'T ", "'t ")
        t = re.sub(r'(?<=\s)(And|Or|Of|The|To|In|On|For|A|An|At|By|With|Vs|From|Into)(?=\s)', lambda m: m.group(1).lower(), t)
    return t
def _ok(t): return t and 4 <= len(t) <= 90 and not SKIP.match(t)
def from_docx(path):
    d = docx.Document(path)
    def sty(p):
        try:
            return p.style.name if p.style is not None else ''
        except Exception:
            return ''
    paras = [(sty(p), p.text.strip()) for p in d.paragraphs if p.text.strip()]
    found = {}
    for st, t in paras:
        m = re.match(r'^chapter\s+(\d{1,2}|' + WORDS + r')\b\s*[:.\-–—]?\s*(.*)$', t, re.I)
        if m and len(t) < 120:
            n = _num(m.group(1))
            if n not in found and _ok(_clean(m.group(2))): found[n] = _clean(m.group(2))
    if len(found) >= 5:
        return [found[k] for k in sorted(found)]
    # table of contents lines like 'TITLE<tab>12'
    toc = []
    for st, t in paras:
        m = re.match(r'^(.{4,100}?)\t+\s*\d{1,3}\s*$', t)
        if m:
            c = _clean(m.group(1))
            if _ok(c): toc.append(c)
    toc = list(dict.fromkeys(toc))
    if 6 <= len(toc) <= 80:
        return toc
    for lvl in ('Heading 1', 'Heading 2'):
        hs = [_clean(t) for st, t in paras if st == lvl]
        hs = [h for h in hs if _ok(h)]
        hs = list(dict.fromkeys(hs))
        if 6 <= len(hs) <= 60:
            return hs
    # last resort: short all-bold paragraphs
    bold = []
    for p in d.paragraphs:
        t = p.text.strip()
        try:
            rs = [r for r in p.runs if r.text.strip()]
        except Exception:
            rs = []
        if rs and len(t) < 85 and all(r.bold for r in rs) and not t.endswith(('.', ':')):
            c = _clean(t)
            if _ok(c): bold.append(c)
    bold = list(dict.fromkeys(bold))[1:]
    return bold if len(bold) >= 6 else []
def from_pdf_outline(r):
    out = []
    try: ol = r.outline
    except Exception: ol = []
    for x in ol:
        if isinstance(x, list): continue
        t = _clean(x.title)
        if _ok(t): out.append(t)
    return out
def chapters(pdf_path, docx_path):
    r = pypdf.PdfReader(pdf_path)
    out = from_pdf_outline(r)
    if docx_path and (len(out) < 10 or len(out) > 60):
        d = from_docx(docx_path)
        if d and (len(out) < 6 or len(out) > 60 or len(d) > len(out)): out = d
    return len(r.pages), out

def cut(s, n):
    if len(s) <= n: return s

def clean_name(n):
    n = re.sub(r'^GS-\d+\s+', '', n)

def slug(n):
    return re.sub(r'[^A-Za-z0-9]+', '-', clean_name(n)).strip('-')

def is_calc(n, nav):
    return nav == ['Inputs'] or nav[:1] == ['Inputs'] or re.search(r'Calculator|Planner|Reality Check|Fit Finder|Diagnostic|Quote Builder|Bid', n) is not None

def wrap(d, text, font, width):
    words, lines, cur = text.split(), [], ''
    for w in words:
        t = (cur + ' ' + w).strip()
        if d.textlength(t, font=font) <= width: cur = t
        else: lines.append(cur); cur = w
    if cur: lines.append(cur)
    return lines

def tags_for(name, calc):
    nm = clean_name(name).lower().replace('&', 'and')
    t = []
    words = [w for w in re.findall(r'[a-z0-9]+', nm) if w not in ('and', 'the', 'my', 'do', 'i', 'to', 'this', 'of')]
    if len(nm) <= 20: t.append(nm)
    for i in range(len(words) - 1):
        p = words[i] + ' ' + words[i + 1]
        if len(p) <= 20 and p not in t: t.append(p)
    if calc: t += ['business calculator', 'pricing calculator']
    for g in GENERIC:
        if len(t) >= 13: break
        if g not in t: t.append(g)
    return [x for x in t if 3 <= len(x) <= 20][:13]
